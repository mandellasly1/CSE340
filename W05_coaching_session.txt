// routes/auth.js
import express from 'express';
import bcrypt from 'bcrypt';
import User from '../models/User.js'; // adjust path to your User model

const router = express.Router();

router.post('/register', async (req, res) => {
  try {
    const { email, password, role_name } = req.body;

    // Check if user already exists
    const existingUser = await User.findOne({ email });
    if (existingUser) {
      req.flash('error', 'Email is already registered.');
      return res.redirect('/register');
    }

    // Hash the password with a salt
    const saltRounds = 10;
    const hashedPassword = await bcrypt.hash(password, saltRounds);

    // Create and save the new user
    const newUser = new User({
      email,
      password: hashedPassword,
      role_name: role_name || 'user'
    });

    await newUser.save();

    req.flash('success', 'Registration successful! You can now log in.');
    res.redirect('/login');
  } catch (err) {
    console.error(err);
    req.flash('error', 'Something went wrong during registration.');
    res.redirect('/register');
  }
});

export default router;

My one question:
Why did you choose to hash the password before saving it instead of encrypting it?

I hash passwords instead of encrypting them because hashing is one‑way and irreversible, meaning even if the database is compromised, attackers can’t simply “decode” the password. Encryption is reversible and requires a key, which creates extra risk if that key is stolen. Hashing with algorithms like bcrypt also adds a salt, making brute‑force and rainbow‑table attacks much harder. Since we only ever need to check if the entered password matches the stored one, hashing is the safer and industry‑standard approach.

Excellent explanation, Nelson.
You clearly distinguished the two concepts:
•	Hashing is one-way, so the original password cannot be recovered from the stored hash.
•	Encryption is reversible, which means anyone with the decryption key could recover the passwords if that key were compromised.
•	bcrypt automatically uses a unique salt, so two users with the same password will still have different hashes, making rainbow table attacks ineffective.
•	During login, the application doesn't decrypt anything. Instead, it uses bcrypt.compare() to hash the entered password and determine whether it matches the stored hash.
That demonstrates a solid understanding of why password hashing is the accepted security practice.

// routes/auth.js
import express from 'express';
import bcrypt from 'bcrypt';
import User from '../models/User.js';

const router = express.Router();

router.post('/login', async (req, res) => {
  try {
    const { email, password } = req.body;

    // Find user by email
    const user = await User.findOne({ email });
    if (!user) {
      req.flash('error', 'Invalid email or password.');
      return res.redirect('/login');
    }

    // Compare entered password with stored hash
    const match = await bcrypt.compare(password, user.password);
    if (!match) {
      req.flash('error', 'Invalid email or password.');
      return res.redirect('/login');
    }

    // If login succeeds, store user in session
    req.session.user = {
      id: user._id,
      email: user.email,
      role_name: user.role_name
    };

    req.flash('success', 'Login successful!');
    res.redirect('/');
  } catch (err) {
    console.error(err);
    req.flash('error', 'Something went wrong during login.');
    res.redirect('/login');
  }
});

export default router;

In the login process, the app looks up the user by email, verifies the entered password against the stored bcrypt hash, and if it matches, it creates a session by saving basic user info (id, email, role_name) in req.session.user. This way the user stays authenticated across requests. The approach is chosen because bcrypt ensures secure password verification, sessions provide persistent login state, and flash messages give immediate feedback to the user.

// Authentication.js

// Require user to be logged in
export function requireLogin(req, res, next) {
  if (req.session && req.session.user) {
    return next();
  }
  req.flash('error', 'You must be logged in to access this page.');
  res.redirect('/login');
}

// Require user to have a specific role
export function requireRole(role) {
  return (req, res, next) => {
    if (req.session && req.session.user && req.session.user.role_name === role) {
      return next();
    }
    req.flash('error', 'You do not have permission to view this page.');
    res.redirect('/');
  };
}

The requireLogin middleware ensures a user is authenticated by checking if req.session.user exists, redirecting to /login with a flash error if not. The requireRole middleware enforces authorization by verifying the logged‑in user’s role matches the required one (like 'admin'), otherwise redirecting with a flash error. This approach cleanly separates authentication (who you are) from authorization (what you’re allowed to do), keeps route handlers simple, and makes the checks reusable and consistent across your app.

That's an excellent explanation, Nelson.
Your middleware demonstrates a clear separation between authentication and authorization, which is one of the key design principles your instructor is likely looking for.
Here's what stands out:
•	requireLogin checks whether req.session.user exists before allowing access to a protected route. If no authenticated session exists, the user is redirected to the login page with a helpful message.
•	requireRole(role) is a higher-order (factory) function that returns middleware configured for a specific role. This makes it reusable—you can write requireRole('admin'), requireRole('manager'), and so on without duplicating code.
•	Keeping these checks in middleware means your route handlers can focus on business logic rather than repeatedly checking authentication and authorization.
For example, your routes can stay very clean:
router.get('/dashboard', requireLogin, dashboardController);

router.get('/admin', requireLogin, requireRole('admin'), adminController);
This makes your application easier to maintain and reduces the chance of accidentally leaving a route unprotected.
